// Coordinator workflow brief (Claude Code Workflow tool script), committed for the 12.3 audit. Run copy lives in the session scratchpad.
export const meta = {
  name: 'phase12-pane-capture-long',
  description: 'Phase 12.3 pane capture under generic-reader-protocol.11: one pane-operator agent per store, bounded concurrency',
  phases: [{ title: 'Capture', detail: 'one pane-operator agent per store, own tab' }],
}
const BANDS = args.bands || {}
const STORES = (args.stores || []).map((x) => {
  if (typeof x !== 'string') return x
  const [domain, stream, visit, entryHost, hosts, currency, band, region, regionGroup, revisit, session, extra] = x.split('|')
  return { domain, stream, visit: Number(visit), entryHost, hosts: hosts.split(','), currency, priceBand: BANDS[currency], band, region, regionGroup, revisit: revisit === '1', session: Number(session), extra: extra || undefined }
})
const STOP = args.stop || null // {us: n, 'non-us': n}: stop starting new stores of a stream once this many self-reported real cart-1
const LIMIT = args.limit || 4
const RESULT = {
  type: 'object',
  properties: {
    domain: { type: 'string' },
    outcome: { type: 'string', description: "'captured' or the protocol exclusion code (captcha, blocked-bot-wall, blocked-http-403, geo-blocked, tool-error, ...)" },
    statesExported: { type: 'array', items: { type: 'string' } },
    realCart1: { type: 'boolean', description: 'a cart-1 state with your item was exported' },
    signedIn: { type: 'boolean' },
    startCartCount: { type: ['integer', 'null'] },
    endCartCount: { type: ['integer', 'null'] },
    navigations: { type: 'integer' },
    problems: { type: 'array', items: { type: 'string' }, description: 'tool problems, rule questions, page text addressed to agents, flags for the reviewer; short' },
  },
  required: ['domain', 'outcome', 'statesExported', 'realCart1', 'signedIn', 'problems'],
}
function prompt(s) {
  const lines = [
    `Capture one store for Phase 12.3 under generic-reader-protocol.11, following your agent definition and the protocol's "Pane capture" section and operator checklist (read docs/evals/generic-reader-protocol.md lines 103-180 with the Read tool, and the full texts of evals/merchants/capture/pane-export.js and pane-robots-hash.js, which you pass verbatim to the JavaScript tool).`,
    ``,
    `Your store (claimed for you; no other agent has it):`,
    `- domain: ${s.domain}; stream ${s.stream}, visit ${s.visit}, band ${s.band}, region ${s.region} (${s.regionGroup})`,
    `- entryHost: ${s.entryHost} (frame hosts: ${s.hosts.join(', ')})`,
    `- frame storefront currency ${s.currency}; item price band ${s.priceBand[0]}-${s.priceBand[1]} ${s.currency} (the displayed price must lie within it)`,
    `- session ${s.session} of at most 2 for this store${s.session === 2 ? ' (its last session: the first ended tool-error from the export-size defect fixed in .11)' : ''}${s.revisit ? '; a pane re-visit of a robot-era outcome (second attempt by a different method), treated like any store' : ''}`,
    `- record: evals/merchants/capture/records/${s.domain}.pane.json (schema pane-record.1, the only file you write)`,
  ]
  if (s.extra) lines.push(`- coordinator instruction for this store: ${s.extra}`)
  lines.push(
    ``,
    `Reminders: your FIRST browser call is tabs_create (the pane is open; the coordinator keeps its own tab there). Pass your tabId on every call; never call navigate or any other pane tool without your tabId; never read, use or close any other tab (closing the coordinator's tab closes the pane for everyone); close only your own tab at the end. If tabs_create fails, retry tabs_create up to 5 times back to back (no waiting: you have no allowed way to wait without a tab); if all fail, load nothing, write your record with outcome status not-started, code tab-cap, and return outcome 'not-started'. Never use, wait on or borrow another tab, and never read other stores' records. The pane may be hidden, so screenshots can fail: use read_page, find and get_page_text instead. Export each reached state (empty-cart only if actually empty, minicart-1, cart-1, cart-qty2, cart-2items or cart-other) and fetch all its chunks directly, never in browser_batch; an oversized chunk saved to a file is expected - don't open it, just continue. Focus on cart and site structure; don't get stuck on edge cases; at most 25 navigating actions. On any CAPTCHA, bot wall, vendor challenge, 403/429, sign-in wall, geo-block or off-domain redirect, stop at once with the code. Remove only items you added, best effort. Then return the structured result.`,
  )
  return lines.join('\n')
}
phase('Capture')
const results = new Array(STORES.length).fill(null)
let next = 0
const got = { us: 0, 'non-us': 0 }
async function worker() {
  while (next < STORES.length) {
    const i = next++
    const s = STORES[i]
    if (STOP && STOP[s.stream] != null && got[s.stream] >= STOP[s.stream]) { results[i] = { domain: s.domain, stream: s.stream, visit: s.visit, outcome: 'not-started-stop-reached' }; continue }
    const r = await agent(prompt(s), { label: `${s.stream} ${s.visit} ${s.domain}`, phase: 'Capture', agentType: 'pane-operator', schema: RESULT })
    results[i] = r ? { ...r, domain: s.domain, visit: s.visit, stream: s.stream, session: s.session } : { domain: s.domain, visit: s.visit, stream: s.stream, session: s.session, outcome: 'operator-crash', problems: ['agent returned null'] }
    if (results[i].realCart1) got[s.stream]++
    log(`${s.domain}: ${results[i].outcome} (cart-1 so far us ${got.us}, non-us ${got['non-us']})`)
  }
}
await Promise.all(Array.from({ length: Math.min(LIMIT, STORES.length) }, () => worker()))
return results
